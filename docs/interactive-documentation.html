🔐 Authentication

TeachAI uses JWT (JSON Web Tokens) for authentication. All protected endpoints require a valid bearer token.

Getting Started

POST /api/auth/register POST

Register a new user account

{ "name": "John Doe", "email": "john.doe@example.com", "password": "securePassword123", "role": "student" }
POST /api/auth/login POST

Authenticate and receive access token

{ "email": "john.doe@example.com", "password": "securePassword123" }

Using Authentication Tokens

Include the JWT token in the Authorization header for all protected requests:

curl -H "Authorization: Bearer YOUR_JWT_TOKEN" \ -H "Content-Type: application/json" \ http://localhost:5000/api/auth/profile

Token Management

  • Tokens expire after 24 hours by default
  • Refresh tokens are automatically issued for seamless authentication
  • Store tokens securely (localStorage, sessionStorage, or secure cookies)
  • Handle token expiration with automatic refresh or redirect to login

💡 Code Examples

Ready-to-use code examples in multiple programming languages.

JavaScript/Node.js

const axios = require('axios'); // Authentication async function loginUser(email, password) { try { const response = await axios.post('http://localhost:5000/api/auth/login', { email: email, password: password }); const token = response.data.data.token; localStorage.setItem('auth_token', token); return token; } catch (error) { console.error('Login failed:', error.response.data); throw error; } } // Send chat message async function sendChatMessage(message, token) { try { const response = await axios.post('http://localhost:5000/api/chat/send', { message: message, subject: 'general', grade: '10' }, { headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' } }); return response.data.data.response; } catch (error) { console.error('Chat request failed:', error.response.data); throw error; } } // Generate quiz async function generateQuiz(topic, gradeLevel) { try { const response = await axios.post('http://localhost:8000/quiz/generate', { topic: topic, grade_level: gradeLevel, num_questions: 5, question_types: ['multiple_choice', 'true_false'], difficulty: 'medium' }); return response.data.quiz; } catch (error) { console.error('Quiz generation failed:', error.response.data); throw error; } }

Python

import requests import json class TeachAIClient: def __init__(self, backend_url="http://localhost:5000/api", ai_url="http://localhost:8000"): self.backend_url = backend_url self.ai_url = ai_url self.token = None def login(self, email, password): """Authenticate user and store token""" response = requests.post(f"{self.backend_url}/auth/login", json={"email": email, "password": password}) response.raise_for_status() self.token = response.json()["data"]["token"] return self.token def send_chat_message(self, message, subject="general", grade="10"): """Send message to AI chat""" headers = {"Authorization": f"Bearer {self.token}"} data = { "message": message, "subject": subject, "grade": grade } response = requests.post(f"{self.backend_url}/chat/send", json=data, headers=headers) response.raise_for_status() return response.json()["data"]["response"] def generate_quiz(self, topic, grade_level="10th grade", num_questions=5): """Generate AI-powered quiz""" data = { "topic": topic, "grade_level": grade_level, "num_questions": num_questions, "question_types": ["multiple_choice", "true_false"], "difficulty": "medium" } response = requests.post(f"{self.ai_url}/quiz/generate", json=data) response.raise_for_status() return response.json()["quiz"] def grade_essay(self, essay_text, prompt, grade_level="high_school"): """Grade essay using AI""" data = { "essay_text": essay_text, "prompt": prompt, "grade_level": grade_level, "rubric_criteria": [ "thesis_clarity", "evidence_support", "organization", "grammar_mechanics" ], "max_score": 100 } response = requests.post(f"{self.ai_url}/essay/grade", json=data) response.raise_for_status() return response.json() # Usage example client = TeachAIClient() token = client.login("user@example.com", "password123") # Send chat message response = client.send_chat_message("Explain photosynthesis") print(f"AI Response: {response}") # Generate quiz quiz = client.generate_quiz("Cell Biology", "9th grade", 10) print(f"Generated quiz with {len(quiz['questions'])} questions")

SDK Quick Start - JavaScript

import { TeachAIClient } from '@teachai/sdk'; const client = new TeachAIClient({ backendUrl: 'http://localhost:5000/api', aiServiceUrl: 'http://localhost:8000', apiKey: 'your-api-key' // Optional for development }); // Authenticate await client.auth.login('user@example.com', 'password123'); // Send chat message const response = await client.chat.sendMessage({ message: 'Explain photosynthesis', subject: 'biology', grade: '10' }); // Generate quiz const quiz = await client.quiz.generate({ topic: 'Cell Biology', gradeLevel: '10th grade', numQuestions: 5, difficulty: 'medium' }); // Grade essay const grading = await client.essay.grade({ text: 'Student essay content...', prompt: 'Write about climate change', gradeLevel: 'high_school' });
